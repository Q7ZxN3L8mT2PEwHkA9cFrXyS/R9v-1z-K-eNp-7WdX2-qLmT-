local function LoadWords()
    if loaded then return end
    pcall(function()
        local res = request({Url = url, Method = "GET"})
        if res and res.Success and res.Body then
            for w in res.Body:gmatch("[^\r\n]+") do
                local wordLower = w:lower()
                table.insert(Words, wordLower)
                WordSet[wordLower] = true
                local firstLetter = wordLower:sub(1,1)

                if not WordDictionary[firstLetter] then
                    WordDictionary[firstLetter] = {}
                end

                table.insert(WordDictionary[firstLetter], wordLower)
            end

            table.sort(Words)

            for _, list in pairs(WordDictionary) do
                table.sort(list)
            end

            loaded = true
        end
    end)
end

local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
end

local function formatWord(word, prefix, index)
    local prefixLen = #prefix
    local firstPart = word:sub(1, prefixLen)
    local restPart = word:sub(prefixLen + 1)

    local prefixColor

    if index == 1 then
        prefixColor = "0,255,0"
    elseif index == 2 then
        prefixColor = "255,255,0"
    else
        prefixColor = "255,0,0"
    end

    return string.format(
        '<font color="rgb(%s)">%s</font>%s',
        prefixColor,
        firstPart,
        restPart
    )
end

local function SuggestWords(input, count)
    if not loaded then
        return {"loading words...", "please wait"}
    end

    if #Words == 0 then
        return {"no words available", "check connection"}
    end

    input = input:lower()

    local cacheKey = input.."_"..count.."_"..sortMode

    if sortMode ~= "Random" and sortMode ~= "Killer" and searchCache[cacheKey] then
        return searchCache[cacheKey]
    end

    local results = {}
    local firstLetter = input:sub(1,1)
    local wordList = WordDictionary[firstLetter] or Words
    local possible = {}

    for i = 1, #wordList do
        local word = wordList[i]

        if word:sub(1,#input) == input then
            table.insert(possible, word)
        end
    end

    if sortMode == "Shortest" then

        table.sort(possible, function(a,b)
            return #a < #b
        end)

    elseif sortMode == "Longest" then

        table.sort(possible, function(a,b)
            return #a > #b
        end)

    elseif sortMode == "Random" then

        shuffle(possible)

    elseif sortMode == "Killer" then

        local link1Words = {}
        local link2Words = {}
        local link3Words = {}
        local normalWords = {}

        for i = 1,#possible do
            local word = possible[i]

            if KillerMap1[word] then
                table.insert(link1Words, word)

            elseif KillerMap2[word] then
                table.insert(link2Words, word)

            elseif KillerMap3[word] then
                table.insert(link3Words, word)

            else
                table.insert(normalWords, word)
            end
        end

        shuffle(link1Words)
        shuffle(link2Words)
        shuffle(link3Words)
        shuffle(normalWords)

        possible = {}

        local limit = count

        local target1 = math.floor(limit * 0.50)
        local target2 = math.floor(limit * 0.30)
        local target3 = math.floor(limit * 0.20)

        local function append(source, amount)
            for i = 1, math.min(amount, #source) do
                table.insert(possible, source[i])
            end
        end

        append(link1Words, target1)
        append(link2Words, target2)
        append(link3Words, target3)

        for i = 1,#normalWords do
            table.insert(possible, normalWords[i])
        end
    end

    local maxResults = math.min(count,#possible)

    for i = 1,maxResults do
        table.insert(results, possible[i])
    end

    if sortMode ~= "Random" and sortMode ~= "Killer" then
        if not searchCache[cacheKey] then
            cacheSize += 1
        end

        searchCache[cacheKey] = results

        if cacheSize > MAX_CACHE then
            searchCache = {}
            cacheSize = 0
        end
    end

    return results
end

local currentResults = {}

function UpdateSuggestions(fromTyping)
    if not loaded then return end
    if not h then return end
    local text = h.Text or ""

    if fromTyping then
        currentPage = 1
        if #text < 1 then
            ClearSuggestions()
            pageLabel.Text = "Page 0/0"
            prevButton.Visible = false
            nextButton.Visible = false
            return
        end
        currentResults = SuggestWords(text, 1000)
        if autoTypeEnabled and #currentResults > 0 then
            autoTypeTarget = currentResults[1]
        else
            autoTypeTarget = nil
        end
    end

    ClearSuggestions()

    local totalPages = math.max(1, math.ceil(#currentResults/wordsPerPage))
    if currentPage > totalPages then currentPage = totalPages end
    if currentPage < 1 then currentPage = 1 end

    pageLabel.Text = "Page "..currentPage.."/"..totalPages
    prevButton.Visible = currentPage > 1
    nextButton.Visible = currentPage < totalPages

    local startIndex = (currentPage-1)*wordsPerPage + 1
    local endIndex = math.min(currentPage*wordsPerPage, #currentResults)
    for i=startIndex,endIndex do
        local word = currentResults[i]
        local btn = Instance.new("TextButton", list)
        btn.Size = UDim2.new(1,0,0,22)
        btn.BackgroundColor3 = Color3.fromRGB(45,45,45)
        btn.TextColor3 = Color3.fromRGB(255,255,255)
        btn.Font = Enum.Font.GothamBold
        btn.TextSize = 12
        btn.RichText = true
        local displayWord = formatWord(
            word,
            h.Text:lower(),
            i - startIndex + 1
        )

        if UsedWords[word] then
            displayWord = displayWord .. " (already used)"
        end

btn.Text = displayWord
        btn.AutoButtonColor = true
        Instance.new("UICorner", btn).CornerRadius = UDim.new(0,4)
        btn.Selectable = false
        btn.MouseButton1Click:Connect(function()
            h.Text = word
            UpdateSuggestions(true)
        end)
    end
end

local function detectPrefix()
    if currentWordGui and currentWordGui.Parent then
        local obj = currentWordGui
        local letters = {}

        for _, child in ipairs(obj:GetChildren()) do
            if child and child:IsA("GuiObject") and child.Visible then
                local letterLabel = child:FindFirstChild("Letter")

                if letterLabel and letterLabel:IsA("TextLabel") then
                    table.insert(letters, {
                        letterLabel.Text,
                        child.AbsolutePosition.X
                    })
                end
            end
        end

        table.sort(letters, function(a, b)
            return a[2] < b[2]
        end)

        local result = ""

        for i = 1, #letters do
            result = result .. letters[i][1]
        end

        return string.lower(result)
    end

    if not currentWordGui or not currentWordGui.Parent then
        currentWordGui = PlayerGui:FindFirstChild("CurrentWord", true)
    end

    return ""
end

local function GetAutoTypeTarget(prefix)
if not prefix or prefix == "" then
return nil
end

prefix = prefix:lower()

local firstLetter = prefix:sub(1, 1)
local bucket = WordDictionary[firstLetter] or Words

if not bucket then
    return nil
end

local candidates = {}

for i = 1, #bucket do
    local word = bucket[i]

    if word:sub(1, #prefix) == prefix
    and #word > #prefix
    and not UsedWords[word] then
        table.insert(candidates, word)
    end
end

if #candidates == 0 then
    return nil
end

return candidates[math.random(1, #candidates)]

end

local VirtualInputManager = game:GetService("VirtualInputManager")

local function SendCharacter(char)
    if not char or char == "" then
        return
    end

    char = char:lower()

    if char == "-" or char:match("^[a-z]$") then
        local success = pcall(function()
            VirtualInputManager:SendTextInput(char)
        end)

        if success then
            return
        end

        if char == "-" then
            if type(keypress) == "function" then
                keypress(0xBD)
                task.wait(0.02)

                if type(keyrelease) == "function" then
                    keyrelease(0xBD)
                end
            end
            return
        end

        local keyCode = Enum.KeyCode[string.upper(char)]

        if keyCode and type(keypress) == "function" then
            local keyMap = {
                a = 0x41, b = 0x42, c = 0x43,
                d = 0x44, e = 0x45, f = 0x46,
                g = 0x47, h = 0x48, i = 0x49,
                j = 0x4A, k = 0x4B, l = 0x4C,
                m = 0x4D, n = 0x4E, o = 0x4F,
                p = 0x50, q = 0x51, r = 0x52,
                s = 0x53, t = 0x54, u = 0x55,
                v = 0x56, w = 0x57, x = 0x58,
                y = 0x59, z = 0x5A
            }

            local code = keyMap[char]

            if code then
                keypress(code)
                task.wait(0.02)

                if type(keyrelease) == "function" then
                    keyrelease(code)
                end
            end
        end

        return
    end

    pcall(function()
        VirtualInputManager:SendTextInput(char)
    end)
end

local function GetCharacterDelay()
    -- Standard WPM calculation:
    -- 1 word = 5 characters
    -- seconds per character = 60 / (WPM * 5)

    return 60 / (currentWPM * 5)
end

local function AutoTypeWord(word, startingPrefix, generation)
    if not word or word == "" then
        return false
    end

    if not autoTypeEnabled then
        return false
    end

    if generation ~= autoTypeGeneration then
        return false
    end

    word = word:lower()
    startingPrefix = startingPrefix:lower()

    if word:sub(1, #startingPrefix) ~= startingPrefix then
        return false
    end

    local remaining = word:sub(#startingPrefix + 1)

    for i = 1, #remaining do
        if not autoTypeEnabled then
            return false
        end

        if generation ~= autoTypeGeneration then
            return false
        end

        local currentPrefix = detectPrefix():lower()
        local expectedPrefix = word:sub(1, #startingPrefix + i - 1)

        if currentPrefix ~= startingPrefix
        and currentPrefix ~= expectedPrefix then
            return false
        end

        SendCharacter(remaining:sub(i, i))

        task.wait(GetCharacterDelay())
    end

    local checkTimeout = 0

    while autoTypeEnabled
    and generation == autoTypeGeneration
    and checkTimeout < 20 do

        local finalPrefix = detectPrefix():lower()

        if finalPrefix == word then
            break
        end

        task.wait(0.05)
        checkTimeout += 1
    end

    if detectPrefix():lower() ~= word then
        return false
    end

    if type(keypress) == "function" then
        keypress(0x0D)
        task.wait(0.02)

        if type(keyrelease) == "function" then
            keyrelease(0x0D)
        end
    else
        VirtualInputManager:SendKeyEvent(
            true,
            Enum.KeyCode.Return,
            false,
            game
        )

        task.wait(0.02)

        VirtualInputManager:SendKeyEvent(
            false,
            Enum.KeyCode.Return,
            false,
            game
        )
    end

    task.wait(0.25)

    return true
end

task.spawn(function()
    while true do
        if autoTypeEnabled and not autoTypeRunning then
            local prefix = detectPrefix()

            if #prefix >= 1
            and not string.find(prefix, "%.%.%.")
            and not string.find(prefix, "#+") then

                local target = GetAutoTypeTarget(prefix)

                if target
                and target ~= ""
                and #target > #prefix
                and target:sub(1, #prefix) == prefix then

                    autoTypeRunning = true

                    local generation = autoTypeGeneration
                    autoTypeTarget = target

                    local success = AutoTypeWord(
                        target,
                        prefix,
                        generation
                    )

                    if success
                    and autoTypeEnabled
                    and generation == autoTypeGeneration then
                        MarkWordUsed(target)
                    end

                    local oldPrefix = prefix
                    local timeout = 0

                    while autoTypeEnabled
                    and generation == autoTypeGeneration
                    and timeout < 100 do

                        local newPrefix = detectPrefix()

                        if newPrefix ~= ""
                        and newPrefix ~= oldPrefix
                        and not string.find(newPrefix, "%.%.%.")
                        and not string.find(newPrefix, "#+") then
                            break
                        end

                        task.wait(0.05)
                        timeout += 1
                    end

                    autoTypeTarget = nil
                    autoTypeRunning = false
                end
            end
        end

        task.wait(0.05)
    end
end)
